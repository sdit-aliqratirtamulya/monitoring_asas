Integrasi Backend Google Apps Script (GAS) & Google Sheets

Konfigurasi URL Web App GAS agar data dari HP pengawas tersimpan ke Google Sheets

Akses file Drive: Siapa saja yang memiliki link

Jadwal Rolling Pengawas: Atur rotasi pengawas harian & per sesi di tab Jadwal_Pengawas pada Google Sheets (kolom: Hari_Tanggal, Sesi, Ruang, ID_Pengawas, Nama_Pengawas). Ruang tugas pengawas akan otomatis berpindah sesuai hari & sesi yang dipilih!

  1. Buka Google Sheets baru, lalu pilih menu Ekstensi → Apps Script.
  2. Salin kode di bawah ini ke editor Code.gs, lalu klik Deploy → New deployment → Web app (Execute as: Me, Who has access: Anyone).
  3. Salin URL Web App yang dihasilkan dan tempelkan pada kolom URL Web App Google Apps Script di atas.
/**
 * BACKEND GOOGLE APPS SCRIPT (Code.gs) - SD IT AL-IQRA
 * Endpoint Two-Way Sync (doGet & doPost) untuk SPA Monitoring ASAS
 */
function setupSheetsIfNeeded() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var struktur = {
    "Pengaturan_Sekolah": ["Kunci", "Nilai", "Keterangan"],
    "Master_Pengawas": ["ID_Akun", "Role", "Nama_Pengawas", "Jabatan", "Ruang_Tugas", "PIN", "Link_Foto_Drive"],
    "Master_Peserta": ["No_Peserta", "Nama_Siswa", "Kelas", "Ruang"],
    "Presensi_ASAS": ["Waktu_Sync", "Hari_Tanggal", "Sesi_Mapel", "No_Peserta", "Nama_Siswa", "Kelas", "Ruang", "Status", "Keterangan"],
    "Berita_Acara": ["ID_BA", "Waktu_Lapor", "Hari_Tanggal", "Ruang", "Nama_Pengawas", "Kondisi_Amplop", "Catatan_Kejadian", "Status_Pengesahan", "Waktu_Pengesahan"],
    "Presensi_Pengawas": ["ID_Log", "Hari_Tanggal", "Sesi_Mapel", "Ruang", "Nama_Pengawas", "Waktu_CheckIn", "Sumber"]
  };

  Object.keys(struktur).forEach(function(namaSheet) {
    var sh = ss.getSheetByName(namaSheet);
    if (!sh) {
      sh = ss.insertSheet(namaSheet);
      sh.appendRow(struktur[namaSheet]);
      sh.getRange(1, 1, 1, struktur[namaSheet].length).setFontWeight("bold").setBackground("#d1fae5");
      if (namaSheet === "Pengaturan_Sekolah") {
        sh.appendRow(["LINK_LOGO_DRIVE", "", "Tempel URL Google Drive Logo SD IT AL-IQRA"]);
        sh.appendRow(["TARIF_HONOR_HARI", "25000", "Nominal Honor Pengawas per Hari (Rp)"]);
        sh.appendRow(["TARIF_KONSUMSI_HARI", "15000", "Nominal Konsumsi Pengawas per Hari (Rp)"]);
      }
      if (namaSheet === "Master_Pengawas") {
        sh.appendRow(["op_aliqra", "operator", "Operator Sekolah SD IT AL-IQRA", "Operator Dapodik & Admin ASAS", "SEMUA", "1234", ""]);
        sh.appendRow(["ks_aliqra", "kepsek", "Kepala Sekolah SD IT AL-IQRA", "Penanggung Jawab ASAS", "SEMUA", "1234", ""]);
      }
    }
  });
  return ss;
}

function doGet(e) {
  var ss = setupSheetsIfNeeded();
  var sheetMasterPeserta = ss.getSheetByName("Master_Peserta");
  var sheetPresensi = ss.getSheetByName("Presensi_ASAS");
  var sheetBA = ss.getSheetByName("Berita_Acara");
  var sheetPengawas = ss.getSheetByName("Presensi_Pengawas");
  var sheetMasterPengawas = ss.getSheetByName("Master_Pengawas");
  var sheetPengaturan = ss.getSheetByName("Pengaturan_Sekolah");

  var hasil = {
    status: "success",
    pulledAt: new Date().toISOString(),
    logoDriveUrl: "",
    tarifHonorPerHari: 25000,
    tarifKonsumsiPerHari: 15000,
    masterPengawas: [],
    masterPeserta: [],
    presensi: [],
    beritaAcara: [],
    presensiPengawas: []
  };

  if (sheetPengaturan && sheetPengaturan.getLastRow() > 1) {
    var rowsSet = sheetPengaturan.getDataRange().getValues();
    for (var s = 1; s < rowsSet.length; s++) {
      var k = String(rowsSet[s][0] || "").trim();
      var v = String(rowsSet[s][1] || "").trim();
      if (k === "LINK_LOGO_DRIVE") hasil.logoDriveUrl = v;
      if (k === "TARIF_HONOR_HARI" && v) hasil.tarifHonorPerHari = Number(v) || 25000;
      if (k === "TARIF_KONSUMSI_HARI" && v) hasil.tarifKonsumsiPerHari = Number(v) || 15000;
    }
  }

  if (sheetMasterPengawas && sheetMasterPengawas.getLastRow() > 1) {
    var rowsMP = sheetMasterPengawas.getDataRange().getValues();
    for (var m = 1; m < rowsMP.length; m++) {
      if (!rowsMP[m][0] && !rowsMP[m][2]) continue;
      hasil.masterPengawas.push({
        id: String(rowsMP[m][0] || ("akun_" + m)).trim(),
        role: String(rowsMP[m][1] || "pengawas").trim().toLowerCase(),
        nama: String(rowsMP[m][2] || "").trim(),
        jabatan: String(rowsMP[m][3] || "").trim(),
        ruangTugas: String(rowsMP[m][4] || "Ruang 01").trim(),
        pin: String(rowsMP[m][5] || "1234").trim(),
        fotoDriveUrl: String(rowsMP[m][6] || "").trim()
      });
    }
  }

  if (sheetMasterPeserta && sheetMasterPeserta.getLastRow() > 1) {
    var rowsSiswa = sheetMasterPeserta.getDataRange().getValues();
    for (var p = 1; p < rowsSiswa.length; p++) {
      if (!rowsSiswa[p][0] && !rowsSiswa[p][1]) continue;
      hasil.masterPeserta.push({
        id: p,
        noUjian: String(rowsSiswa[p][0] || "").trim(),
        nama: String(rowsSiswa[p][1] || "").trim(),
        kelas: String(rowsSiswa[p][2] || "").trim(),
        ruang: String(rowsSiswa[p][3] || "Ruang 01").trim()
      });
    }
  }

  if (sheetPresensi && sheetPresensi.getLastRow() > 1) {
    var rowsP = sheetPresensi.getDataRange().getValues();
    for (var i = 1; i < rowsP.length; i++) {
      hasil.presensi.push({
        hari: String(rowsP[i][1] || "").trim(),
        mapel: String(rowsP[i][2] || "").trim(),
        noUjian: String(rowsP[i][3] || "").trim(),
        nama: String(rowsP[i][4] || "").trim(),
        kelas: String(rowsP[i][5] || "").trim(),
        ruang: String(rowsP[i][6] || "").trim(),
        status: String(rowsP[i][7] || "H").trim(),
        keterangan: String(rowsP[i][8] || "").trim()
      });
    }
  }

  if (sheetBA && sheetBA.getLastRow() > 1) {
    var rowsBA = sheetBA.getDataRange().getValues();
    for (var j = 1; j < rowsBA.length; j++) {
      hasil.beritaAcara.push({
        id: Number(rowsBA[j][0]) || (Date.now() + j),
        waktu: String(rowsBA[j][1] || "").trim(),
        hari: String(rowsBA[j][2] || "").trim(),
        ruang: String(rowsBA[j][3] || "").trim(),
        pengawas: String(rowsBA[j][4] || "").trim(),
        amplop: String(rowsBA[j][5] || "").trim(),
        catatan: String(rowsBA[j][6] || "").trim(),
        disahkanKepsek: String(rowsBA[j][7] || "").trim() === "SAH",
        waktuPengesahan: String(rowsBA[j][8] || "").trim()
      });
    }
  }

  if (sheetPengawas && sheetPengawas.getLastRow() > 1) {
    var rowsPW = sheetPengawas.getDataRange().getValues();
    for (var k = 1; k < rowsPW.length; k++) {
      hasil.presensiPengawas.push({
        id: Number(rowsPW[k][0]) || (Date.now() + k),
        hari: String(rowsPW[k][1] || "").trim(),
        mapel: String(rowsPW[k][2] || "").trim(),
        ruang: String(rowsPW[k][3] || "").trim(),
        namaPengawas: String(rowsPW[k][4] || "").trim(),
        waktuCheckIn: String(rowsPW[k][5] || "").trim(),
        sumber: String(rowsPW[k][6] || "Google Sheets").trim()
      });
    }
  }

  return ContentService
    .createTextOutput(JSON.stringify(hasil))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    var ss = setupSheetsIfNeeded();
    var data = JSON.parse(e.postData.contents);
    var waktuSync = Utilities.formatDate(new Date(), "Asia/Jakarta", "dd/MM/yyyy HH:mm:ss");

    // 1. Update Pengaturan Sekolah
    var sheetSet = ss.getSheetByName("Pengaturan_Sekolah");
    if (sheetSet) {
      sheetSet.clearContents();
      sheetSet.appendRow(["Kunci", "Nilai", "Keterangan"]);
      sheetSet.appendRow(["LINK_LOGO_DRIVE", data.logoDriveUrl || "", "Tempel URL Google Drive Logo SD IT AL-IQRA"]);
      sheetSet.appendRow(["TARIF_HONOR_HARI", data.tarifHonorPerHari || 25000, "Nominal Honor Pengawas per Hari (Rp)"]);
      sheetSet.appendRow(["TARIF_KONSUMSI_HARI", data.tarifKonsumsiPerHari || 15000, "Nominal Konsumsi Pengawas per Hari (Rp)"]);
    }

    // 2. Sinkron Master Peserta
    if (Array.isArray(data.masterPeserta) && data.masterPeserta.length > 0) {
      var sheetMPes = ss.getSheetByName("Master_Peserta");
      sheetMPes.clearContents();
      sheetMPes.appendRow(["No_Peserta", "Nama_Siswa", "Kelas", "Ruang"]);
      var rowsMPes = data.masterPeserta.map(function(s) {
        return [s.noUjian, s.nama, s.kelas, s.ruang];
      });
      sheetMPes.getRange(2, 1, rowsMPes.length, 4).setValues(rowsMPes);
    }

    // 3. Upsert Presensi Harian per Hari + Mapel + No_Peserta
    if (Array.isArray(data.peserta) && data.peserta.length > 0) {
      var sheetPres = ss.getSheetByName("Presensi_ASAS");
      var existing = sheetPres.getDataRange().getValues();
      var indexMap = {};
      for (var r = 1; r < existing.length; r++) {
        var key = existing[r][1] + "__" + existing[r][2] + "__" + existing[r][3];
        indexMap[key] = r + 1;
      }
      data.peserta.forEach(function(p) {
        var k = (data.hari || "") + "__" + (data.mapel || "") + "__" + p.noUjian;
        var rowVals = [waktuSync, data.hari || "", data.mapel || "", p.noUjian, p.nama, p.kelas, p.ruang, p.status || "H", p.keterangan || ""];
        if (indexMap[k]) {
          sheetPres.getRange(indexMap[k], 1, 1, 9).setValues([rowVals]);
        } else {
          sheetPres.appendRow(rowVals);
        }
      });
    }

    // 4. Upsert Berita Acara
    if (Array.isArray(data.beritaAcara) && data.beritaAcara.length > 0) {
      var sheetBA = ss.getSheetByName("Berita_Acara");
      var exBA = sheetBA.getDataRange().getValues();
      var baMap = {};
      for (var b = 1; b < exBA.length; b++) {
        baMap[String(exBA[b][0])] = b + 1;
      }
      data.beritaAcara.forEach(function(item) {
        var rowBA = [
          item.id,
          item.waktu,
          item.hari,
          item.ruang,
          item.pengawas,
          item.amplop,
          item.catatan,
          item.disahkanKepsek ? "SAH" : "MENUNGGU",
          item.waktuPengesahan || ""
        ];
        if (baMap[String(item.id)]) {
          sheetBA.getRange(baMap[String(item.id)], 1, 1, 9).setValues([rowBA]);
        } else {
          sheetBA.appendRow(rowBA);
        }
      });
    }

    // 5. Upsert Presensi Pengawas
    if (Array.isArray(data.presensiPengawas) && data.presensiPengawas.length > 0) {
      var sheetPW = ss.getSheetByName("Presensi_Pengawas");
      var exPW = sheetPW.getDataRange().getValues();
      var pwSet = {};
      for (var w = 1; w < exPW.length; w++) {
        pwSet[exPW[w][1] + "__" + exPW[w][3] + "__" + exPW[w][4]] = true;
      }
      data.presensiPengawas.forEach(function(pw) {
        var keyPW = pw.hari + "__" + pw.ruang + "__" + pw.namaPengawas;
        if (!pwSet[keyPW]) {
          sheetPW.appendRow([pw.id, pw.hari, pw.mapel, pw.ruang, pw.namaPengawas, pw.waktuCheckIn, pw.sumber]);
          pwSet[keyPW] = true;
        }
      });
    }

    return ContentService
      .createTextOutput(JSON.stringify({ status: "success", syncedAt: waktuSync }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: "error", message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
